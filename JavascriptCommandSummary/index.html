Loop Templates

1. For Loop


for (let i = 0; i < limit; i++) {
  // Code to execute on each iteration
  // Access the current index with 'i'
}

//example:

for (let i = 0; i < array.length; i++) {
  console.log(array[i]);
}

2. While Loop


while (condition) {
  // Code to execute as long as condition is true
  // Remember to update the condition within the loop to avoid an infinite loop
}

Example:

  console.log(counter);

}

3. Do...while Loop

} while (condition);

example:

4. For...of Loop

  // Code to execute for each element in iterable
}

Example:

  console.log(value);
}

5. For...in Loop


for (const key in object) {
  if (object.hasOwnProperty(key)) {
    // Code to execute for each key in object
    // Access the key with 'key' and the value with 'object[key]'
  }
}

Example:

  console.log(key + ": " + person[key]);
}

6. Nested Loop


for (let i = 0; i < outerLimit; i++) {
  for (let j = 0; j < innerLimit; j++) {
    // Code to execute within the nested loop
    // Access outer loop index with 'i' and inner loop index with 'j'
  }
}

Example:

  [4, 5, 6],
  [7, 8, 9]

  }
}

Async HTML Function Templates

1. Basic Async Function with Await


async function fetchData() {
  try {
    const response = await fetch("https://api.example.com/data");
    const data = await response.json();
    console.log(data);
  } catch (error) {
    console.error("Error fetching data:", error);
  }
}

2. Fetch Request with Async/Await


async function fetchDataFromAPI(url) {
  try {
    const response = await fetch(url); // Initiates the HTTP request
    if (!response.ok) throw new Error("Network response was not ok");
    
    const data = await response.json(); // Waits for the data to be parsed as JSON
    return data; // Returns the fetched data
  } catch (error) {
    console.error("Error fetching data:", error);
    throw error; // Re-throws error to handle it at the call site if needed
  }
}

// Usage example
fetchDataFromAPI("https://api.example.com/data")
  .then(data => console.log("Data received:", data))
  .catch(error => console.error("Fetch error:", error));

3. Asynchronous Function with setTimeout


function delayedOperation(duration) {
  return new Promise(resolve => {
    setTimeout(() => {
      resolve("Operation completed");
    }, duration);
  });
}

// Usage in an async function
async function performAsyncOperation() {
  console.log("Starting operation...");
  const result = await delayedOperation(2000); // Waits for 2 seconds
  console.log(result); // Logs "Operation completed"
}

performAsyncOperation();

5. Running Multiple Asynchronous Operations in Parallel (Promise.all)


async function fetchMultipleData() {
  try {
    const [data1, data2] = await Promise.all([
      fetch("https://api.example.com/data1").then(res => res.json()),
      fetch("https://api.example.com/data2").then(res => res.json())
    ]);
    console.log("Data 1:", data1);
    console.log("Data 2:", data2);
  } catch (error) {
    console.error("Error fetching multiple data:", error);
  }
}

fetchMultipleData();

6. Asynchronous Function with setInterval (i.e. for Polling)


function startPolling(interval) {
  const intervalId = setInterval(async () => {
    try {
      const response = await fetch("https://api.example.com/status");
      const status = await response.json();
      console.log("Current status:", status);
      
      if (status.complete) {
        clearInterval(intervalId); // Stop polling if the condition is met
        console.log("Polling stopped - Operation complete.");
      }
    } catch (error) {
      console.error("Polling error:", error);
      clearInterval(intervalId); // Stop polling in case of error
    }
  }, interval);
}

// Start polling every 5 seconds
startPolling(5000);

Standard Array Structures

1. Loop Through an Array (forEach)


        let arr = [1, 2, 3, 4, 5];
arr.forEach(element => {
  console.log(element);  // Perform some action with each element
});

2. Map an Array (Create a New Array Based on Transformation)


        let arr = [1, 2, 3, 4, 5];
let doubled = arr.map(element => element * 2);  // [2, 4, 6, 8, 10]

3. Filter an Array (Create a Subset of Elements)


        let arr = [1, 2, 3, 4, 5];
let evenNumbers = arr.filter(element => element % 2 === 0);  // [2, 4]

4. Reduce an Array (Accumulate Values into a Single Output)


        let arr = [1, 2, 3, 4, 5];
let sum = arr.reduce((accumulator, element) => accumulator + element, 0);  // 15

5. Find an Element in an Array (find)


        let arr = [1, 2, 3, 4, 5];
let firstEven = arr.find(element => element % 2 === 0);  // 2

6. Find the Index of an Element (indexOf / findIndex)


        let arr = [1, 2, 3, 4, 5];
let index = arr.indexOf(3);  // 2 (returns -1 if not found)

let indexOfFirstEven = arr.findIndex(element => element % 2 === 0);  // 1

7. Check if an Array Includes an Element (includes)


        let arr = [1, 2, 3, 4, 5];
let hasThree = arr.includes(3);  // true
let hasSeven = arr.includes(7);  // false

8. Sort an Array (sort)


        let arr = [3, 1, 4, 1, 5, 9];
arr.sort((a, b) => a - b);  // [1, 1, 3, 4, 5, 9] (ascending order)

arr.sort((a, b) => b - a);  // [9, 5, 4, 3, 1, 1] (descending order)

9. Add an Element to the End of an Array (push)


        let arr = [1, 2, 3];
arr.push(4);  // [1, 2, 3, 4]

10. Remove the Last Element from an Array (pop)


        let arr = [1, 2, 3];
let lastElement = arr.pop();  // 3; arr becomes [1, 2]

11. Add an Element to the Beginning of an Array (unshift)


        let arr = [1, 2, 3];
arr.unshift(0);  // [0, 1, 2, 3]

12. Remove the First Element from an Array (shift)


        let arr = [1, 2, 3];
let firstElement = arr.shift();  // 1; arr becomes [2, 3]

13. Slice a Portion of an Array (slice)


        let arr = [1, 2, 3, 4, 5];
let slicedArr = arr.slice(1, 3);  // [2, 3] (does not include index 3)

14. Splice an Array (Add, Remove, or Replace Elements)


        let arr = [1, 2, 3, 4, 5];
arr.splice(2, 1);          // Removes 1 element at index 2; arr becomes [1, 2, 4, 5]
arr.splice(2, 0, 99);      // Adds 99 at index 2; arr becomes [1, 2, 99, 4, 5]
arr.splice(2, 1, 100);     // Replaces 1 element at index 2 with 100; arr becomes [1, 2, 100, 4, 5]

15. Flatten a Nested Array (flat)


        let arr = [1, [2, [3, 4]], 5];
let flattened = arr.flat(2);  // [1, 2, 3, 4, 5]

16. Check if All Elements Meet a Condition (every)


        let arr = [1, 2, 3, 4, 5];
let allPositive = arr.every(element => element > 0);  // true

17. Check if Some Elements Meet a Condition (some)


        let arr = [1, 2, 3, 4, 5];
let hasEvenNumber = arr.some(element => element % 2 === 0);  // true

18. Concatenate Arrays (concat)


        let arr1 = [1, 2, 3];
let arr2 = [4, 5, 6];
let combined = arr1.concat(arr2);  // [1, 2, 3, 4, 5, 6]

19. Convert an Array to a String (join)


        let arr = [1, 2, 3];
let joinedString = arr.join("-");  // "1-2-3"

20. Convert a String to an Array (split)
